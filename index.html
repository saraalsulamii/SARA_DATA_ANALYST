-- Calculate Monthly Revenue and MoM Growth %
WITH MonthlySales AS (
    SELECT 
        YEAR(SaleDate) AS SalesYear,
        MONTH(SaleDate) AS SalesMonth,
        SUM(TotalAmount) AS CurrentRevenue
    FROM Sales
    GROUP BY YEAR(SaleDate), MONTH(SaleDate)
)
SELECT 
    SalesYear, SalesMonth, CurrentRevenue,
    LAG(CurrentRevenue, 1) OVER (ORDER BY SalesYear, SalesMonth) AS PriorRevenue,
    ROUND(((CurrentRevenue - LAG(CurrentRevenue, 1) OVER (ORDER BY SalesYear, SalesMonth)) 
    / NULLIF(LAG(CurrentRevenue, 1) OVER (ORDER BY SalesYear, SalesMonth), 0)) * 100, 2) AS MoM_Growth_PCT
FROM MonthlySales;
              

-- Customer Spending Segmentation
SELECT 
    c.CustomerID, c.CustomerName,
    SUM(s.TotalAmount) AS TotalSpent,
    CASE 
        WHEN SUM(s.TotalAmount) >= 10000 THEN 'VIP Tier'
        WHEN SUM(s.TotalAmount) BETWEEN 4000 AND 9999 THEN 'Regular'
        ELSE 'Low Engagement'
    END AS CustomerSegment
FROM Customers c
JOIN Sales s ON c.CustomerID = s.CustomerID
GROUP BY c.CustomerID, c.CustomerName;

Nashville Housing Data Cleaning Script

-- 1. Standardize Sale Date
ALTER TABLE NashvilleHousing ADD SaleDateConverted Date;
UPDATE NashvilleHousing SET SaleDateConverted = CONVERT(Date, SaleDate);

-- 2. Populate Missing Property Addresses
UPDATE a
SET PropertyAddress = ISNULL(a.PropertyAddress, b.PropertyAddress)
FROM PortfolioProject.dbo.NashvilleHousing a
JOIN PortfolioProject.dbo.NashvilleHousing b
    ON a.ParcelID = b.ParcelID
    AND a.[UniqueID ] <> b.[UniqueID ]
WHERE a.PropertyAddress IS NULL;

-- 3. Remove Duplicate Records
WITH RowNumCTE AS (
    SELECT *,
        ROW_NUMBER() OVER (
            PARTITION BY ParcelID, PropertyAddress, SalePrice, SaleDate, LegalReference
            ORDER BY UniqueID
        ) row_num
    FROM PortfolioProject.dbo.NashvilleHousing
)
DELETE FROM RowNumCTE WHERE row_num > 1;

Retail Analytics & RFM Script

-- 1. Revenue & Profitability per Category
SELECT 
    p.Category,
    COUNT(DISTINCT s.TransactionID) AS Total_Orders,
    SUM(s.Quantity * p.UnitPrice) AS Total_Revenue,
    SUM(s.Quantity * (p.UnitPrice - p.CostPrice)) AS Total_Profit
FROM Sales s
JOIN Products p ON s.ProductID = p.ProductID
GROUP BY p.Category
ORDER BY Total_Revenue DESC;

-- 2. Customer RFM Segmentation
SELECT 
    c.CustomerID, c.CustomerName,
    SUM(s.TotalAmount) AS TotalSpent,
    CASE 
        WHEN SUM(s.TotalAmount) >= 10000 THEN 'VIP Tier'
        WHEN SUM(s.TotalAmount) BETWEEN 4000 AND 9999 THEN 'Regular'
        ELSE 'Low Engagement'
    END AS CustomerSegment
FROM Customers c
JOIN Sales s ON c.CustomerID = s.CustomerID
GROUP BY c.CustomerID, c.CustomerName;
